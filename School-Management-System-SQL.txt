DROP DATABASE IF EXISTS altamayoz_sis;
-- 1. Create the central school database
CREATE DATABASE altamayoz_sis;

-- 2. Switch to the school database environment
USE altamayoz_sis;

-- 3. Create the STUDENT table with all required fields
CREATE TABLE STUDENT(
   ID int PRIMARY KEY AUTO_INCREMENT,
   full_name VARCHAR(255),
   date_of_birth DATE,
   gender CHAR(1), -- Stores 'M' or 'F' as required by the project specifications
   enrollment_date DATE,
   email VARCHAR(100) UNIQUE,
   grade_level VARCHAR(40),
   track VARCHAR(50),
   gpa DECIMAL(5,2) -- Configured to support cumulative GPA scales up to 100.00
);

-- 4. Create the TEACHER table to store faculty members' data
CREATE TABLE TEACHER(
   ID int PRIMARY KEY AUTO_INCREMENT,
   full_name VARCHAR(255),
   date_of_birth DATE,
   gender CHAR(1), -- Stores 'M' or 'F' to remain consistent with student records
   email VARCHAR(100) UNIQUE,
   office_number VARCHAR(20)
);

-- 5. Create the COURSE table to register available school subjects
CREATE TABLE COURSE(
   ID int PRIMARY KEY AUTO_INCREMENT, -- Using standardized ID notation as per assignment notes
   course_name VARCHAR(100)
);

-- 6. Display all created tables inside the database to verify their existence
